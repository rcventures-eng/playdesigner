Feature B: Fix Backspace Behavior in Notes
Problem: When typing in a note with the cursor active, pressing backspace deletes the ENTIRE note instead of just the last character. This makes editing notes impossible.
Root Cause: The backspace key event is being captured at the canvas/app level and triggering "delete selected object" instead of being handled by the text input.
Fix Required:
Implement proper state management for note editing:
StateSingle ClickDouble ClickBackspace BehaviorNote not selectedSelects note (shows handles)Activates edit mode (cursor)Does nothingNote selected (no cursor)Already selectedActivates edit modeDeletes entire noteNote in edit mode (cursor visible)N/AN/ADeletes last character (normal typing)
Implementation:
typescript// Track note states
type NoteState = 'unselected' | 'selected' | 'editing';

const [selectedNoteId, setSelectedNoteId] = useState<string | null>(null);
const [editingNoteId, setEditingNoteId] = useState<string | null>(null);

// Handle note click
const handleNoteClick = (noteId: string, e: React.MouseEvent) => {
  e.stopPropagation();
  
  if (editingNoteId === noteId) {
    // Already editing, do nothing (let text input handle clicks)
    return;
  }
  
  // Single click - select the note (for moving/deleting)
  setSelectedNoteId(noteId);
  setEditingNoteId(null); // Exit edit mode if was editing another note
};

// Handle note double-click
const handleNoteDoubleClick = (noteId: string, e: React.MouseEvent) => {
  e.stopPropagation();
  
  // Double click - enter edit mode (show cursor)
  setSelectedNoteId(noteId);
  setEditingNoteId(noteId);
  
  // Focus the text input
  setTimeout(() => {
    const input = document.querySelector(`#note-input-${noteId}`) as HTMLInputElement;
    input?.focus();
  }, 0);
};

// Handle keydown at app/canvas level
const handleKeyDown = (e: KeyboardEvent) => {
  if (e.key === 'Backspace') {
    // If we're editing a note, let the input handle backspace normally
    if (editingNoteId) {
      // DO NOT prevent default - let the input delete characters
      return;
    }
    
    // If a note is selected (but not editing), delete the entire note
    if (selectedNoteId) {
      e.preventDefault();
      deleteNote(selectedNoteId);
      setSelectedNoteId(null);
    }
  }
  
  // Escape exits edit mode
  if (e.key === 'Escape') {
    setEditingNoteId(null);
  }
};

// Click outside note to deselect
const handleCanvasClick = (e: React.MouseEvent) => {
  // Only if clicking on canvas, not on a note
  if ((e.target as HTMLElement).classList.contains('canvas')) {
    setSelectedNoteId(null);
    setEditingNoteId(null);
  }
};
Note Component:
tsxconst PlayNote = ({ note, isSelected, isEditing, onClick, onDoubleClick, onChange }) => {
  const inputRef = useRef<HTMLInputElement>(null);
  
  useEffect(() => {
    if (isEditing && inputRef.current) {
      inputRef.current.focus();
      // Place cursor at end of text
      inputRef.current.setSelectionRange(note.text.length, note.text.length);
    }
  }, [isEditing]);
  
  return (
    <div
      className={`play-note ${isSelected ? 'selected' : ''} ${isEditing ? 'editing' : ''}`}
      onClick={(e) => onClick(note.id, e)}
      onDoubleClick={(e) => onDoubleClick(note.id, e)}
      style={{ left: note.x, top: note.y }}
    >
      {isEditing ? (
        <input
          ref={inputRef}
          id={`note-input-${note.id}`}
          type="text"
          value={note.text}
          onChange={(e) => onChange(note.id, e.target.value)}
          onKeyDown={(e) => {
            // Stop propagation so canvas doesn't capture backspace
            e.stopPropagation();
          }}
          className="note-text-input"
        />
      ) : (
        <span className="note-text">{note.text}</span>
      )}
      
      {isSelected && !isEditing && (
        <div className="note-handles">
          {/* Resize/move handles */}
        </div>
      )}
    </div>
  );
};

Part 2: Delete Logic (Cascade Rules)
Two delete scenarios with different behaviors:
Delete FromWhat HappensPlay LibraryDeletes from Library AND automatically removes from ALL Team PlaybooksTeam PlaybookRemoves from that playbook ONLY, Play Library copy remains
Implementation:
typescript// Delete from Play Library (MASTER delete)
const deleteFromLibrary = async (playId: string) => {
  // This deletes the play entirely
  await fetch(`/api/plays/${playId}`, { method: 'DELETE' });
  
  // The backend should cascade delete from all playbook references
  // OR do it client-side:
  await fetch(`/api/playbook-entries?playId=${playId}`, { method: 'DELETE' });
  
  // Refresh both library and playbooks
  queryClient.invalidateQueries(['plays']);
  queryClient.invalidateQueries(['playbooks']);
};

// Delete from Team Playbook (remove reference only)
const deleteFromPlaybook = async (playbookId: string, playId: string) => {
  // This only removes the play from THIS playbook
  // The play still exists in the library
  await fetch(`/api/playbooks/${playbookId}/plays/${playId}`, { method: 'DELETE' });
  
  // Refresh playbook
  queryClient.invalidateQueries(['playbook', playbookId]);
};
Backend (if not already implemented):
typescript// DELETE /api/plays/:id - Master delete
app.delete('/api/plays/:id', async (req, res) => {
  const { id } = req.params;
  
  // Delete from all playbook entries first (cascade)
  await db.delete(playbookEntries).where(eq(playbookEntries.playId, id));
  
  // Then delete the play itself
  await db.delete(plays).where(eq(plays.id, id));
  
  res.json({ success: true });
});

// DELETE /api/playbooks/:playbookId/plays/:playId - Remove from playbook only
app.delete('/api/playbooks/:playbookId/plays/:playId', async (req, res) => {
  const { playbookId, playId } = req.params;
  
  // Only delete the playbook entry, not the play itself
  await db.delete(playbookEntries)
    .where(
      and(
        eq(playbookEntries.playbookId, playbookId),
        eq(playbookEntries.playId, playId)
      )
    );
  
  res.json({ success: true });
});
