Overview

Blank Page (existing functionality)

All page types should render cleanly on 8.5" x 11" format for Google Docs export.

When "Roster Page" is selected, pull data from the team's existing roster (if available) or create a template:
typescriptconst createRosterPage = async (playbookId: string) => {
  // Get team associated with this playbook
  const playbook = await getPlaybook(playbookId);
  const teamId = playbook.teamId;
  
  // Fetch existing roster data if available
  let players: RosterPlayer[] = [];
  let coaches: RosterCoach[] = [];
  
  if (teamId) {
    const team = await getTeam(teamId);
    players = team.players || [];
    coaches = team.coaches || [];
  }
  
  // If no existing data, create template placeholders
  if (players.length === 0) {
    players = [
      { id: '1', number: '', name: 'Player Name', position: 'Position', notes: '' },
      { id: '2', number: '', name: 'Player Name', position: 'Position', notes: '' },
      { id: '3', number: '', name: 'Player Name', position: 'Position', notes: '' },
    ];
  }
  
  if (coaches.length === 0) {
    coaches = [
      { id: '1', name: 'Coach Name', role: 'Head Coach', phone: '', email: '' },
      { id: '2', name: 'Coach Name', role: 'Assistant Coach', phone: '', email: '' },
    ];
  }
  
  // Get next position in playbook
  const entries = await getPlaybookEntries(playbookId);
  const nextPosition = entries.length;
  
  // Create the roster page entry
  const rosterPage = await db.insert(playbookEntries).values({
    id: generateId(),
    playbookId,
    playId: null,
    type: 'roster',
    title: 'Team Roster',
    position: nextPosition,
    rosterData: JSON.stringify({ players, coaches }),
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString()
  }).returning();
  
  return rosterPage[0];
};

typescriptinterface SplitsPage {
  id: string;

interface TeamSplits {
  // Formation splits
  formations: FormationSplit[];
  
  // Situation splits
  situations: SituationSplit[];
  
  // Personnel splits
  personnel: PersonnelSplit[];
  
  // Custom splits
  custom: CustomSplit[];
}

interface FormationSplit {
  id: string;
  name: string;              // "Spread", "I-Form", "Trips Right"
  runPlays: number;
  passPlays: number;
  percentage: number;
}

interface SituationSplit {
  id: string;
  situation: string;         // "1st & 10", "3rd & Long", "Red Zone"
  runPlays: number;
  passPlays: number;
  totalPlays: number;
}

interface PersonnelSplit {
  id: string;
  personnel: string;         // "11 Personnel", "12 Personnel", "21 Personnel"
  usage: number;             // Percentage
  avgYards: number;
}

interface CustomSplit {
  id: string;
  label: string;
  value: string;
}
Auto-Generation Logic
typescriptconst createSplitsPage = async (playbookId: string) => {
  // Get playbook entries to analyze
  const entries = await getPlaybookEntries(playbookId);
  const plays = entries.filter(e => e.type === 'play').map(e => e.play);
  
  // Calculate formation splits from existing plays
  const formationCounts = {};
  plays.forEach(play => {
    const formation = play.formation || 'Unknown';
    if (!formationCounts[formation]) {
      formationCounts[formation] = { run: 0, pass: 0 };
    }
    if (play.concept === 'run') {
      formationCounts[formation].run++;
    } else {
      formationCounts[formation].pass++;
    }
  });
  
  const formations = Object.entries(formationCounts).map(([name, counts]: [string, any]) => ({
    id: generateId(),
    name,
    runPlays: counts.run,
    passPlays: counts.pass,
    percentage: Math.round(((counts.run + counts.pass) / plays.length) * 100)
  }));
  
  // Calculate situation splits
  const situationCounts = {};
  plays.forEach(play => {
    (play.situations || []).forEach(situation => {
      if (!situationCounts[situation]) {
        situationCounts[situation] = { run: 0, pass: 0 };
      }
      if (play.concept === 'run') {
        situationCounts[situation].run++;
      } else {
        situationCounts[situation].pass++;
      }
    });
  });
  
  const situations = Object.entries(situationCounts).map(([situation, counts]: [string, any]) => ({
    id: generateId(),
    situation,
    runPlays: counts.run,
    passPlays: counts.pass,
    totalPlays: counts.run + counts.pass
  }));
  
  // Create template if no data
  if (formations.length === 0) {
    formations.push(
      { id: '1', name: 'Spread', runPlays: 0, passPlays: 0, percentage: 0 },
      { id: '2', name: 'Trips Right', runPlays: 0, passPlays: 0, percentage: 0 },
      { id: '3', name: 'I-Form', runPlays: 0, passPlays: 0, percentage: 0 }
    );
  }
  
  if (situations.length === 0) {
    situations.push(
      { id: '1', situation: '1st & 10', runPlays: 0, passPlays: 0, totalPlays: 0 },
      { id: '2', situation: '3rd & Long', runPlays: 0, passPlays: 0, totalPlays: 0 },
      { id: '3', situation: 'Red Zone', runPlays: 0, passPlays: 0, totalPlays: 0 }
    );
  }
  
  const splitsData: TeamSplits = {
    formations,
    situations,
    personnel: [],
    custom: []
  };
  
  // Get next position
  const allEntries = await getPlaybookEntries(playbookId);
  const nextPosition = allEntries.length;
  
  // Create the splits page entry
  const splitsPage = await db.insert(playbookEntries).values({
    id: generateId(),
    playbookId,
    playId: null,
    type: 'splits',
    title: 'Team Splits',
    position: nextPosition,
    splitsData: JSON.stringify(splitsData),
    createdAt: new Date().toISOString(),
    updatedAt: new Date().toISOString()
  }).returning();
  
  return splitsPage[0];
};

Document Generation for All Page Types
typescriptconst generatePlaybookDocument = async (playbookId: string) => {
  const entries = await getPlaybookEntries(playbookId);
  const playbook = await getPlaybook(playbookId);
  
  // Create document with 8.5 x 11 page size
  const doc = new PDFDocument({
    size: 'LETTER', // 8.5" x 11"
    margins: {
      top: 72,      // 1 inch
      bottom: 72,
      left: 72,
      right: 72
    }
  });
  
  // Title page
  doc.fontSize(28).text(playbook.name, { align: 'center' });
  doc.moveDown(2);
  doc.fontSize(14).text(playbook.teamName || 'Team Playbook', { align: 'center' });
  doc.addPage();
  
  // Generate each page by type
  for (const entry of entries) {
    switch (entry.type) {
      case 'play':
        await renderPlayPage(doc, entry.play);
        break;
      case 'blank':
        renderBlankPage(doc, entry);
        break;
      case 'roster':
        renderRosterPage(doc, entry);
        break;
      case 'splits':
        renderSplitsPage(doc, entry);
        break;
    }
    
    // Add new page after each entry (except last)
    if (entries.indexOf(entry) < entries.length - 1) {
      doc.addPage();
    }
  }
  
  return doc;
};
