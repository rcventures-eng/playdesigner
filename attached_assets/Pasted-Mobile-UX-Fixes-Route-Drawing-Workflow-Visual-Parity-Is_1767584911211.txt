```
## Issue C: Redesign Route Drawing Workflow

**Problem:** The current workflow is confusing. Need a clearer flow where selections are made, then user explicitly starts drawing.

**New Workflow for PASS Routes:**
1. Long press player → Menu opens
2. Select "Pass Route" → Highlights orange, menu stays open
3. Select "Straight" or "Curved" → Highlights orange
4. (Optional) Check "Motion" and/or "Primary" checkboxes
5. Tap "Draw Route" button → Menu closes, drawing mode activates

**New Workflow for RUN or BLOCK:**
1. Long press player → Menu opens
2. Select "Run Route" or "Block" → Immediately starts drawing (no additional steps)

**New Menu Layout:**
```
┌─────────────────────────────────────────┐

├─────────────────────────────────────────┤

│ └─────────┘ └─────────┘ └─────────┘    │
├─────────────────────────────────────────┤
│ Route Style (only visible if Pass)      │
│ ┌───────────────┐ ┌───────────────┐    │
│ │   Straight    │ │    Curved     │    │
│ └───────────────┘ └───────────────┘    │
├─────────────────────────────────────────┤

Implementation:
tsxconst PlayerActionMenu = ({ player, onClose, onStartDrawing }) => {
  const [routeType, setRouteType] = useState<'pass' | 'run' | 'block' | null>(null);
  const [style, setStyle] = useState<'straight' | 'curved'>('straight');
  const [isMotion, setIsMotion] = useState(false);
  const [isPrimary, setIsPrimary] = useState(false);

  const handleRouteTypeSelect = (type: 'pass' | 'run' | 'block') => {
    setRouteType(type);
    
    // For Run or Block, immediately start drawing
    if (type === 'run' || type === 'block') {
      onStartDrawing({
        playerId: player.id,
        routeType: type,
        style: 'straight', // Default for run/block
        isMotion: false,
        isPrimary: false
      });
      onClose();
    }
    // For Pass, keep menu open for additional options
  };

  const handleDrawRoute = () => {
    onStartDrawing({
      playerId: player.id,
      routeType: 'pass',
      style,
      isMotion,
      isPrimary
    });
    onClose();
  };

  return (
    <div className="player-menu">
      <div className="menu-header">
        <span>Player {player.label}</span>
        <button onClick={onClose}>✕</button>
      </div>

      <div className="menu-section">
        <label>Select Action</label>
        <div className="menu-options">
          <button 
            className={`menu-option ${routeType === 'pass' ? 'selected' : ''}`}
            onClick={() => handleRouteTypeSelect('pass')}
          >
            Pass Route
          </button>
          <button 
            className={`menu-option ${routeType === 'run' ? 'selected' : ''}`}
            onClick={() => handleRouteTypeSelect('run')}
          >
            Run Route
          </button>
          <button 
            className={`menu-option ${routeType === 'block' ? 'selected' : ''}`}
            onClick={() => handleRouteTypeSelect('block')}
          >
            Block
          </button>
        </div>
      </div>

      {/* Only show style options for Pass routes */}
      {routeType === 'pass' && (
        <>
          <div className="menu-section">
            <label>Route Style</label>
            <div className="menu-options">
              <button 
                className={`menu-option ${style === 'straight' ? 'selected' : ''}`}
                onClick={() => setStyle('straight')}
              >
                Straight
              </button>
              <button 
                className={`menu-option ${style === 'curved' ? 'selected' : ''}`}
                onClick={() => setStyle('curved')}
              >
                Curved
              </button>
            </div>
          </div>

          <div className="menu-section">
            <label>Options</label>
            <div className="checkbox-row">
              <label className="checkbox-option">
                <input 
                  type="checkbox" 
                  checked={isMotion} 
                  onChange={(e) => setIsMotion(e.target.checked)} 
                />
                <span>Motion</span>
              </label>
              <label className="checkbox-option">
                <input 
                  type="checkbox" 
                  checked={isPrimary} 
                  onChange={(e) => setIsPrimary(e.target.checked)} 
                />
                <span>Primary</span>
              </label>
            </div>
          </div>

          <button className="draw-route-cta" onClick={handleDrawRoute}>
            Draw Route
          </button>
        </>
      )}
    </div>
  );
};
